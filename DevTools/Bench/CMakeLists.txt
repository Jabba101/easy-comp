cmake_minimum_required(VERSION 3.22)
project(Bench2 VERSION 1.0.0)
add_subdirectory(/home/claude/deps/juce-9.0.4 juce)
if(NOT DEFINED SRC)
    set(SRC /home/claude/EasyPlugins)
endif()
set(P ${SRC})
set(SHARED ${P}/Shared/EasyProcessor.cpp ${P}/Shared/EasyWindow.cpp ${P}/Shared/Look.cpp ${P}/Shared/PresetBrowser.cpp ${P}/Shared/Widgets.cpp)
foreach(tool CompBench TransientBench)
    juce_add_console_app(${tool} PRODUCT_NAME "${tool}")
    target_compile_definitions(${tool} PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
    target_include_directories(${tool} PRIVATE ${P}/Shared ${P}/Effects)
    target_link_libraries(${tool} PRIVATE juce::juce_audio_utils juce::juce_audio_formats PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_lto_flags)
endforeach()
target_sources(CompBench PRIVATE CompBench.cpp ${P}/Plugins/EasyComp/PluginProcessor.cpp ${P}/Plugins/EasyComp/PluginEditor.cpp ${SHARED})
target_include_directories(CompBench PRIVATE ${P}/Plugins/EasyComp)
target_compile_definitions(CompBench PRIVATE JucePlugin_Name="Easy Comp")
target_sources(TransientBench PRIVATE TransientBench.cpp ${P}/Plugins/EasyTransient/PluginProcessor.cpp ${P}/Plugins/EasyTransient/PluginEditor.cpp ${SHARED})
target_include_directories(TransientBench PRIVATE ${P}/Plugins/EasyTransient)
target_compile_definitions(TransientBench PRIVATE JucePlugin_Name="Easy Transient")
